# SpideyBot runtime dependencies.
#
# Core set only. The /bypass feature additionally needs camoufox (which pins its
# own Playwright); that is deliberately installed later, as it pulls 100 MB+ of
# browser binaries that nothing else uses.

# ── Telegram ────────────────────────────────────────────────────────
# Kurigram is a Pyrogram fork; the import name is still `pyrogram`.
kurigram>=2.2.26
# The AES accelerator, installed explicitly rather than through kurigram's `[fast]`
# extra: that extra pins upstream `tgcrypto`, whose newest wheel is cp311 and whose
# plain-C sdist therefore has to be compiled on 3.14. `tgcrypto-pyrofork` is that same
# C extension re-released with cp314 wheels, so no toolchain is needed on any platform
# the project runs on. It installs the module name `tgcrypto`, which is the name
# `pyrogram.crypto.aes` imports, and it keeps the bytearray/in-place CTR state contract
# that Kurigram's obfuscated2 transports and CDN downloads depend on — a rewrite that
# takes immutable bytes there imports fine and then fails at runtime instead.
tgcrypto-pyrofork>=1.2.8
# `[fast]` was also the only source of the fast event loop, so dropping the extra has to
# put it back explicitly (D17). uvloop has no Windows build, which is why the split is by
# platform rather than one line: uvloop on Linux (0.22.1 is the first release with cp314
# wheels, which the project's Python floor requires) and winuvloop on Windows, where it
# selects winloop. Both are optimisations, so neither is a required capability.
uvloop>=0.22.1; sys_platform == "linux"
winuvloop>=0.2.5; sys_platform == "win32"

# ── Configuration ───────────────────────────────────────────────────
pydantic-settings>=2.7
structlog>=24.1

# ── Database (PostgreSQL, async) ────────────────────────────────────
SQLAlchemy[asyncio]>=2.0
asyncpg>=0.29
alembic>=1.13
# Bundles a PostgreSQL distribution so the bot runs with no Docker and no
# installer. The wheel is py3-none-win_amd64 / py3-none-manylinux, i.e.
# Python-version agnostic.
postgresql-binaries>=18.6

# ── HTTP ────────────────────────────────────────────────────────────
curl_cffi>=0.7

# ── HTML scraping ───────────────────────────────────────────────────
# The web-bridge adapters (downloader/sites/*) read a third-party saver site's HTML answer, which a
# regex cannot do reliably. Declared rather than relied on transitively: it arrives today only
# because an engine depends on it, so an engine upgrade could take it away silently.
beautifulsoup4>=4.12

# ── Web-bridge crypto ───────────────────────────────────────────────
# The same adapters also decrypt: AES-GCM in downloader/sites/bysejikuar.py, AES-CBC in
# mega.py. Declared here because the runtime image installs this file and not pyproject.toml,
# where it would otherwise be the only declared dependency — without it `main.py run` cannot
# even import the downloader, while `main.py check` still passes.
cryptography>=50.0.2

# ── Media extractors ────────────────────────────────────────────────
yt-dlp>=2025.1.1
gallery-dl>=1.29
cyberdrop-dl-patched>=10.0

# ── Reddit ──────────────────────────────────────────────────────────
praw>=7.7

# Test and lint tooling is deliberately not here: it lives in requirements-dev.txt so
# that the runtime image does not ship pytest or ruff (a ~30 MB binary).
